# 01. 構築手順

※Ubuntu 24.04 LTS バージョンのOSで検証済

### pipx hatch and pipenv のインストール

sudo apt-get update

sudo apt-get install -y pipx

pipx install hatch

pipx install pipenv

pipx ensurepath

source \~/.bashrc

 

hatch --version

pipenv --version

### dockerのインストール

sudo apt install -y ca-certificates curl gnupg lsb-release

sudo mkdir -p /etc/apt/keyrings

curl -fsSL  [https://download.docker.com/linux/ubuntu/gpg](https://download.docker.com/linux/ubuntu/gpg)  | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg

echo "deb \[arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg\]  <custom data-type="smartlink" data-id="id-0">https://download.docker.com/linux/ubuntu</custom>   $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

sudo apt update

apt install -y docker-ce docker-ce-cli <custom data-type="smartlink" data-id="id-1">http://containerd.io</custom>  docker-buildx-plugin docker-compose-plugin

sudo systemctl start docker

sudo systemctl enable docker

docker --version

docker compose version

### #Githubのリポジトリをローカルに

cd /home/user/workspace/stpos

git clone https://{github username}:{github key}@github.com/retail-ai-inc/stpos-backend-kugelpos.git

※github keyはパスワードではなく、`ghp_xxxxxxxxxxxxxxxxxxxx` のようなキーとなる
